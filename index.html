#include #include #include #include #include #include // ========== WI-FI CREDENTIALS ========== const char* ssid = "YOUR_WIFI_SSID"; const char* password = "YOUR_WIFI_PASSWORD"; // Host HTTP Web Server on Port 80 WebServer server(80); // ========== PIN SETTINGS ========== const int IR_PIN = 17; // IR Coaster Sensor const int BUZZER_PIN = 16; // Buzzer const int LED_PIN = 4; // NeoPixel DATA INPUT (DIN) // ========== LED SETTINGS ========== const int LED_COUNT = 26; Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800); // ========== LCD SETTINGS ========== LiquidCrystal_I2C lcd(0x27, 16, 2); // Most IR modules output LOW when bottle is resting on coaster const int BOTTLE_PRESENT = LOW; // ========== TIMER SETTINGS ========== const unsigned long DRINK_INTERVAL = 600000UL; // 10 minutes (600,000 ms) const unsigned long RESPONSE_LIMIT = 300000UL; // 5 minutes timeout (300,000 ms) // ========== STATES ========== enum ReminderState { WAITING, WAITING_FOR_REMOVAL, WAITING_FOR_RETURN, ABSENT_PAUSED }; ReminderState state = WAITING; unsigned long drinkTimerStart = 0; unsigned long reminderStart = 0; unsigned long lastLcdUpdate = 0; int drinkStreak = 4; // HydroSip initial streak counter bool isAbsent = false; // ========== SENSOR & LED HELPERS ========== bool bottlePresent() { return digitalRead(IR_PIN) == BOTTLE_PRESENT; } void setBlue() { strip.fill(strip.Color(0, 0, 255)); strip.show(); } void setRed() { strip.fill(strip.Color(255, 0, 0)); strip.show(); } void turnLedsOff() { strip.clear(); strip.show(); } // ========== LCD DISPLAY HELPERS ========== void showWaiting(unsigned long remainingSeconds) { lcd.setCursor(0, 0); lcd.print("HydroBuddy (Ava)"); lcd.setCursor(0, 1); lcd.print("Next: "); lcd.print(remainingSeconds); lcd.print(" sec "); } void startReminder() { reminderStart = millis(); digitalWrite(BUZZER_PIN, HIGH); setRed(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("TIME TO DRINK!"); if (bottlePresent()) { state = WAITING_FOR_REMOVAL; lcd.setCursor(0, 1); lcd.print("Pick up bottle"); } else { state = WAITING_FOR_RETURN; lcd.setCursor(0, 1); lcd.print("Return bottle"); } } void finishReminder() { digitalWrite(BUZZER_PIN, LOW); setBlue(); state = WAITING; drinkTimerStart = millis(); drinkStreak++; lcd.clear(); lcd.setCursor(0, 0); lcd.print("Bottle detected"); lcd.setCursor(0, 1); lcd.print("Streak Updated!"); delay(1200); lcd.clear(); } // ========== REST API HANDLERS FOR HYDROSIP WEB APP ========== // GET /api/status - Sends json data to HydroSip Web Dashboard void handleGetStatus() { server.sendHeader("Access-Control-Allow-Origin", "*"); StaticJsonDocument<300> doc; doc["student"] = "Ava"; doc["coaster_id"] = "AVA-HYDRO-01"; doc["detector_status"] = bottlePresent() ? "DETECTED" : "NOT DETECTED"; doc["streak"] = drinkStreak; doc["buzzer_active"] = (digitalRead(BUZZER_PIN) == HIGH); doc["is_absent"] = isAbsent; unsigned long elapsed = millis() - drinkTimerStart; doc["remaining_seconds"] = (elapsed < DRINK_INTERVAL && !isAbsent) ? ((DRINK_INTERVAL - elapsed) / 1000UL) : 0; String jsonResponse; serializeJson(doc, jsonResponse); server.send(200, "application/json", jsonResponse); } // GET /api/pause - Toggles Absent/Present Standby Mode from Web App void handlePauseToggle() { server.sendHeader("Access-Control-Allow-Origin", "*"); isAbsent = !isAbsent; if (isAbsent) { state = ABSENT_PAUSED; digitalWrite(BUZZER_PIN, LOW); turnLedsOff(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Ava: ABSENT"); lcd.setCursor(0, 1); lcd.print("Coaster Paused"); } else { state = WAITING; drinkTimerStart = millis(); setBlue(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Ava: PRESENT"); lcd.setCursor(0, 1); lcd.print("Coaster Active"); delay(1200); lcd.clear(); } server.send(200, "text/plain", isAbsent ? "PAUSED" : "ACTIVE"); } // OPTIONS CORS Preflight void handleCORS() { server.sendHeader("Access-Control-Allow-Origin", "*"); server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS"); server.sendHeader("Access-Control-Allow-Headers", "Content-Type"); server.send(204); } // ========== SETUP ========== void setup() { Serial.begin(115200); pinMode(IR_PIN, INPUT); pinMode(BUZZER_PIN, OUTPUT); digitalWrite(BUZZER_PIN, LOW); Wire.begin(21, 22); lcd.init(); lcd.backlight(); strip.begin(); strip.setBrightness(40); strip.clear(); strip.show(); setBlue(); lcd.setCursor(0, 0); lcd.print("HydroSip Coaster"); lcd.setCursor(0, 1); lcd.print("Connecting WiFi"); // WiFi Connection with 10-Second Timeout Limit WiFi.mode(WIFI_STA); WiFi.begin(ssid, password); int attempts = 0; while (WiFi.status() != WL_CONNECTED && attempts < 20) { delay(500); Serial.print("."); attempts++; } lcd.clear(); if (WiFi.status() == WL_CONNECTED) { Serial.println("\nWiFi Connected!"); Serial.print("HydroSip IP: "); Serial.println(WiFi.localIP()); server.on("/api/status", HTTP_GET, handleGetStatus); server.on("/api/status", HTTP_OPTIONS, handleCORS); server.on("/api/pause", HTTP_GET, handlePauseToggle); server.on("/api/pause", HTTP_OPTIONS, handleCORS); server.begin(); lcd.setCursor(0, 0); lcd.print("WiFi Connected!"); lcd.setCursor(0, 1); lcd.print(WiFi.localIP()); } else { Serial.println("\nWiFi Offline Mode Started"); lcd.setCursor(0, 0); lcd.print("WiFi: Offline"); lcd.setCursor(0, 1); lcd.print("Coaster Ready!"); } delay(2000); lcd.clear(); drinkTimerStart = millis(); } // ========== MAIN LOOP ========== void loop() { // Handle web server requests from HydroSip Web App server.handleClient(); unsigned long now = millis(); switch (state) { case ABSENT_PAUSED: { // Coaster is on standby while student is absent digitalWrite(BUZZER_PIN, LOW); turnLedsOff(); break; } case WAITING: { setBlue(); unsigned long elapsed = now - drinkTimerStart; if (elapsed >= DRINK_INTERVAL) { startReminder(); break; } unsigned long remaining = (DRINK_INTERVAL - elapsed + 999UL) / 1000UL; if (now - lastLcdUpdate >= 1000UL) { lastLcdUpdate = now; showWaiting(remaining); } break; } case WAITING_FOR_REMOVAL: { digitalWrite(BUZZER_PIN, HIGH); setRed(); if (!bottlePresent()) { state = WAITING_FOR_RETURN; lcd.setCursor(0, 1); lcd.print("Return bottle "); } if (now - reminderStart >= RESPONSE_LIMIT) { digitalWrite(BUZZER_PIN, LOW); setBlue(); state = WAITING; drinkTimerStart = millis(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Reminder ended"); lcd.setCursor(0, 1); lcd.print("Timer restarted"); delay(1200); lcd.clear(); } break; } case WAITING_FOR_RETURN: { digitalWrite(BUZZER_PIN, HIGH); setRed(); if (bottlePresent()) { finishReminder(); break; } if (now - reminderStart >= RESPONSE_LIMIT) { digitalWrite(BUZZER_PIN, LOW); setBlue(); state = WAITING; drinkTimerStart = millis(); lcd.clear(); lcd.setCursor(0, 0); lcd.print("No bottle found"); lcd.setCursor(0, 1); lcd.print("Timer restarted"); delay(1200); lcd.clear(); } break; } } }
